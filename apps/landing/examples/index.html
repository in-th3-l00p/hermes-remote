hermes web — integrating a Hermes agent, end to end

anonymous telegram-style chat

a complete chat webapp against a real Hermes agent — streaming replies, markdown rendering, image attachments, reactions and message edits. the agent never learns who the user is. full source: apps/chat.

1 · run the api next to your agent

# hermes-api proxies your local Hermes agent to the web
npm i -g @in-th3-l00p/hermes-web-api
hermes-api serve --anonymous \
  --cors http://localhost:5173 \
  --upstream http://127.0.0.1:8642 \
  --upstream-key $API_SERVER_KEY

2 · the whole chat is one hook

import { HermesClient, useChat } from "@in-th3-l00p/hermes-web-react";

const client = new HermesClient({ baseUrl: "http://localhost:8643" });

function Chat() {
  const { messages, streaming, error, send, edit, react } =
    useChat({ client });

  return (
    <>
      {messages.map((m) => (
        <Bubble key={m.id} message={m}
          onReact={(emoji) => react(m.id, emoji)}
          onEdit={(text) => edit(m.id, text)} />
      ))}
      <Composer onSend={send} busy={streaming} />
    </>
  );
}

3 · streaming, edits, reactions, attachments

// send() streams — messages update token by token
await send("Explain SSE in one paragraph");

// attach images as data urls; they reach vision models
await send("What is in this photo?", [
  { name: "photo.png", type: "image/png", dataUrl },
]);

// editing a message truncates history and regenerates
await edit(messageId, "Explain WebSockets instead");

// reactions toggle
await react(messageId, "🔥");

plain typescript, no react

import { HermesClient } from "@in-th3-l00p/hermes-web-ts";

const client = new HermesClient({ baseUrl: "http://localhost:8643" });
const session = await client.createSession();

for await (const event of client.sendMessage(session.id, {
  content: "hello!",
})) {
  if (event.event === "delta") process.stdout.write(event.data.text);
}
